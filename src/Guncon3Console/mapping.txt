#guncon buttons and trigger mapping
#format is [device].[command]=[guncommand]
#device can be KEYBOARD, MOUSE (P1), MOUSE2 (P2 aim) or GAMEPAD2 (P2 buttons)
#keyboard command must be the keycode. Run program with [keys] argument to display suported keycodes
#mouse command can be Left, Right, Middle
#guncommand can be Trigger, A1, A2, B1, B2, C1, C2, AClick, BClick, LUp, LDown, LLeft, LRight
#values are CASE SENSITIVE

KEYBOARD.30=C1
KEYBOARD.34=C2

# Player 1 (TetherScript Virtual Mouse Abs)
MOUSE.Left=Trigger
MOUSE.Right=A1
MOUSE.Middle=B2

# Player 2 (TetherScript Virtual Mouse Rel)
MOUSE2.Left=Trigger
MOUSE2.Right=A1
MOUSE2.Middle=B2

# Player 2 buttons as TetherScript Virtual Gamepad (GAMEPAD2.<bit>=<GunButton>)
GAMEPAD2.0=Trigger
GAMEPAD2.1=A1
GAMEPAD2.2=B2

KEYBOARD.37=A2
KEYBOARD.38=B1
KEYBOARD.82=LUp     
KEYBOARD.81=LDown   
KEYBOARD.80=LLeft    
KEYBOARD.79=LRight
